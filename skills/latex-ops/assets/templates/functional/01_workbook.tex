\section{Language focus}
Keep explanations concise and immediately useful for the exercise that follows.

\begin{example}
I work \textbf{in} Toronto. I study \textbf{at} the university.
\end{example}

\begin{exercise}[Controlled practice]
Complete each sentence with the appropriate word.
\begin{enumerate}[leftmargin=*,itemsep=.7em]
  \item I arrive \rule{2cm}{0.4pt} work at 8:00.
  \item She lives \rule{2cm}{0.4pt} Canada.
  \item We meet \rule{2cm}{0.4pt} Monday.
\end{enumerate}
\end{exercise}
